MAIN
enable
configure terminal
hostname Switch0

! Configuración VTP
vtp domain FIUSAC
vtp mode server

! Creación de VLANs
vlan 10
 name ADMIN
vlan 20
 name MERCA
vlan 30
 name VENTAS
exit

! Puertos Troncales Gigabit (hacia ADMIN y MERCA)
interface range GigabitEthernet 0/1 - 2
 switchport mode trunk
 switchport trunk allowed vlan all
 no shutdown
exit

! Puerto Troncal FastEthernet (hacia VENTAS)
interface FastEthernet 0/24
 switchport mode trunk
 switchport trunk allowed vlan all
 no shutdown
exit

end
write memory
